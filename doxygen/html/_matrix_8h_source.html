359template <typename T, std::size_t N, std::size_t M>
360Matrix<T, N, M> Matrix<T, N, M>::Inverse() const requires(N == M) {
361 // TODO(ChristopherStackpole57): requires Determinant to be implemented
362 // static_assert(this->Determinant() != 0);
363
364 // Implements Gauss - Jordan Elimination
365 Matrix<T, N, N> A = *this; // Copy current matrix for easier manipulation
366 Matrix<T, N, N> I = Matrix<T, N, N>::Identity(); // Identity Matrix
367
368 for (std::size_t col = 0; col < N; col++) {
369 // Find Pivot Row
370 std::size_t pivot = col;
371 for (std::size_t row = col + 1; row < N; row++) {
372 if (std::abs(A(row, col)) > std::abs(A(pivot, row))) {
373 pivot = row;
374 }
375 }
376
377 // Check for Singular Matrix
378 if (A(pivot, col) == T{ 0 }) {
379 throw std::runtime_error("AME: Singular Matrices Cannot Be Inverted.");
380 }
381
382 // Swap Current Row with Pivot Row in Both A and I
383 if (pivot != col) {
384 for (std::size_t j = 0; j < N; j++) {
385 std::swap(A(col, j), A(pivot, j));
386 std::swap(I(col, j), I(pivot, j));
387 }
388 }
389
390 // Normalize Pivot Row
391 T pivot_val = A(col, col);
392 for (std::size_t j = 0; j < N; j++) {
393 A(col, j) /= pivot_val;
394 I(col, j) /= pivot_val;
395 }
396
397 // Eliminate All Other Entries in This Column
398 for (std::size_t row = 0; row < N; row++) {
399 if (row == col) continue;
400
401 T factor = A(row, col);
402 for (std::size_t j = 0; j < N; j++) {
403 A(row, j) -= factor * A(col, j);
404 I(row, j) -= factor * I(col, j);
405 }
406 }
407 }
408
409 return I;
410}
